Cases for Kahlua's string library (tools/oracle.sh luastring), run like tests/lua_vm_cases.txt: each block after a
"--@@case" line runs in a fresh environment. Not covered on purpose: a back reference (%1) to a capture slot the
call never opened (the original reads a pooled MatchState left by earlier calls), %f/%e of numbers above 1e300
(the original loops forever), widths of millions of characters.

-- Generated patterns: a fixed linear congruential sequence picks subjects, patterns and replacements; every
-- result (or error message) is printed. No token ends in "%", so no back reference can appear.
local seed = 12345
local function rnd(n) seed = (seed * 1103515245 + 12345) % 2147483648; return seed % n end
local subj = {"a", "b", "c", "1", " ", "(", ")", ".", "A", "%", "ab", "-"}
local toks = {"a", "b", "c", ".", "%a", "%d", "%s", "%w", "%p", "%u", "%l", "%x", "%A", "%S", "[ab]", "[^a]", "[a-c]",
    "[%d%s]", "(", ")", "()", "*", "+", "-", "?", "^", "$", "%b()", "%f[%a]", "%f[%A]", "%.", "1", " ", "[]", "[^]"}
local repls = {"<%0>", "[%1]", "%%", "x", "", "%2%1"}
local function str(...) local n = select("#", ...); local r = {} for i = 1, n do r[i] = tostring((select(i, ...))) end return table.concat(r, ",") end
for i = 1, 2500 do
    local s = {}
    for j = 1, rnd(9) do s[j] = subj[rnd(#subj) + 1] end
    s = table.concat(s)
    local p = {}
    for j = 1, rnd(5) + 1 do p[j] = toks[rnd(#toks) + 1] end
    p = table.concat(p)
    local init = rnd(12) - 4
    local r1 = str(pcall(string.find, s, p, init))
    local r2 = str(pcall(string.match, s, p))
    local r3 = str(pcall(string.gsub, s, p, repls[rnd(#repls) + 1], rnd(4) - 1))
    print(i, s, p, init, (r1:gsub("\n.*", "")), (r2:gsub("\n.*", "")), (r3:gsub("\n.*", "")))
end

-- Generated formats over numbers from a fixed sequence (magnitudes 1e-9 to 1e15, both signs, integers).
local seed = 777
local function rnd(n) seed = (seed * 1103515245 + 12345) % 2147483648; return seed % n end
local convs = {"d", "i", "u", "x", "X", "o", "c", "e", "E", "f", "g", "G", "s", "q"}
local flags = {"", "-", "+", " ", "#", "0", "-0", "+0", " #", "#0"}
for i = 1, 2500 do
    local mag = 10 ^ (rnd(25) - 9)
    local v = (rnd(2000000) / 1000000 - 1) * mag
    if rnd(4) == 0 then v = math.floor(v) end
    if rnd(30) == 0 then v = 0 end
    local spec = "%" .. flags[rnd(#flags) + 1] .. (rnd(3) == 0 and "" or tostring(rnd(21)))
        .. (rnd(3) == 0 and "" or ("." .. tostring(rnd(16)))) .. convs[rnd(#convs) + 1]
    local ok, r = pcall(string.format, spec, v)
    print(i, spec, v, ok, ok and r or (r:gsub("\n.*", "")))
end
